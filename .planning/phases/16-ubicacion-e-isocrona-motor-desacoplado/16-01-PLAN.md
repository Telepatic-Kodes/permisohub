```typescript
import { z } from 'zod'
import { fetchWithTimeout } from '@/lib/scraper'

export function isOrsConfigured(): boolean {
  return !!process.env.ORS_API_KEY
}

export type IsocronaModo = 'caminando' | 'auto'

// Zod schema escrito a partir del payload REAL capturado en Task 2 (scripts/verify-ors-isochrone.mjs)
// — ajusta los nombres de campo de `properties` a lo que realmente devolvió ORS, no a esta plantilla.
const OrsFeatureSchema = z.object({
  type: z.literal('Feature'),
  geometry: z.object({
    type: z.enum(['Polygon', 'MultiPolygon']),
    coordinates: z.array(z.unknown()),
  }),
  properties: z.record(z.string(), z.unknown()),
})

const OrsIsochroneResponseSchema = z.object({
  type: z.literal('FeatureCollection'),
  features: z.array(OrsFeatureSchema),
})

function profileParaModo(modo: IsocronaModo): 'foot-walking' | 'driving-car' {
  return modo === 'caminando' ? 'foot-walking' : 'driving-car'
}

export type ConsultaIsocronaResult =
  | { ok: true; geometria: GeoJSON.Polygon | GeoJSON.MultiPolygon }
  | { ok: false; error: string }

/**
 * Único punto del repo que arma el request body de ORS. `locations` usa
 * [lng, lat] — GeoJSON/ORS, NO [lat, lng] como el filtro `around:` de Overpass
 * (lib/terrenos-ubicacion.ts) ni como los pares lat/lng sueltos de Nominatim
 * (lib/geocoding.ts). Ver Pitfall 2, 16-RESEARCH.md — este codebase ya fue
 * mordido por un bug de orden de ejes equivalente con ArcGIS
 * (app/api/zonificacion/lookup/route.ts línea ~167).
 *
 * Nunca lanza. Retorna {ok:false} ante: sin API key, HTTP no-2xx, timeout,
 * shape inesperado (Zod falla), o 0 features. El caller (obtenerIsocrona() en
 * lib/cabida-comercial-server.ts) es quien decide degradar a
 * circulo_equivalente — este módulo no conoce ese fallback, solo reporta
 * éxito/fracaso de ORS honestamente.
 */
export async function consultarIsocronaOrs(
  lat: number,
  lng: number,
  modo: IsocronaModo,
  minutos: number,
): Promise<ConsultaIsocronaResult> {
  if (!isOrsConfigured()) {
    return { ok: false, error: 'ORS_API_KEY no configurada' }
  }

  const profile = profileParaModo(modo)

  try {
    const res = await fetchWithTimeout(
      `https://api.openrouteservice.org/v2/isochrones/${profile}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json; charset=utf-8',
          'Accept': 'application/geo+json',
          'Authorization': process.env.ORS_API_KEY!, // raw key, NO "Bearer "
        },
        body: JSON.stringify({
          locations: [[lng, lat]], // [lng, lat] — ver comentario de función arriba
          range: [minutos * 60], // segundos
          range_type: 'time',
        }),
      },
      10_000,
    )

    if (!res.ok) {
      return { ok: false, error: `ORS HTTP ${res.status}` }
    }

    const rawJson: unknown = await res.json()
    const parsed = OrsIsochroneResponseSchema.safeParse(rawJson)
    if (!parsed.success) {
      console.error('[isocrona] Respuesta de ORS con forma inesperada:', parsed.error.message)
      return { ok: false, error: 'Respuesta de ORS con formato inesperado' }
    }

    if (parsed.data.features.length === 0) {
      return { ok: false, error: 'ORS no retornó ninguna isócrona para este punto' }
    }

    const geometria = parsed.data.features[0].geometry as GeoJSON.Polygon | GeoJSON.MultiPolygon
    return { ok: true, geometria }
  } catch (err) {
    return { ok: false, error: err instanceof Error ? err.message : 'Error de red al consultar ORS' }
  }
}
```
