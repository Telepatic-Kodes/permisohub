```typescript
import { createServiceClient } from '@/lib/supabase/service'
import { geocodeDireccion, geocodeComunaCentroide } from '@/lib/geocoding'
import type { UbicacionCabida } from '@/lib/cabida-comercial'

interface AtributosRaw {
  locationText?: string
}

/**
 * Resuelve la ubicación de una Oportunidad (mercado_locales_listings) para
 * el motor de cabida comercial. Query MÍNIMA y DEDICADA — no reusa
 * obtenerOportunidadPorId() (ver comentario de cabecera del archivo).
 *
 * Nunca lanza. Retorna null solo si AMBOS geocodings fallan (mismo criterio
 * "non-success explícito, nunca excepción" que geocodeDireccion() ya sigue).
 */
export async function resolverUbicacionDesdeOportunidad(oportunidadId: string): Promise<UbicacionCabida | null> {
  const supabase = createServiceClient()
  const { data: listing, error } = await supabase
    .from('mercado_locales_listings')
    .select('id, titulo, comuna, atributos_raw')
    .eq('id', oportunidadId)
    .maybeSingle()

  if (error || !listing) return null

  const atributos = (listing.atributos_raw ?? {}) as AtributosRaw
  const comuna = listing.comuna as string
  const textoParaGeocodificar = atributos.locationText || (listing.titulo as string)

  return resolverUbicacionDesdeDireccion(textoParaGeocodificar, comuna)
}

/**
 * Resolver genérico por dirección/texto + comuna — la firma que un futuro
 * modo standalone (CABI-03, milestone futuro) va a llamar directamente.
 * En v1.7 nada en la UI la llama todavía salvo el resolver de arriba, pero
 * existe como el seam real (16-RESEARCH.md, Pattern 2) — la duplicación de
 * caminos (oportunidad vs dirección libre) convergiendo acá ES el punto.
 */
export async function resolverUbicacionDesdeDireccion(direccion: string, comuna: string): Promise<UbicacionCabida | null> {
  const geo = await geocodeDireccion(direccion, comuna)
  if (geo.ok && geo.lat !== undefined && geo.lng !== undefined) {
    return {
      lat: geo.lat,
      lng: geo.lng,
      comuna,
      precision: 'aproximada',
      direccionLabel: geo.displayName ?? direccion,
      fuenteTexto: direccion,
    }
  }

  // Fallback explícito por FALLO real (geo.ok === false), nunca por una
  // heurística de "coarse vs fine" — locationText ya es sector-level texto,
  // sin house_number/road que distinguir en la mayoría de los casos (ver
  // 16-RESEARCH.md, Anti-Patterns to Avoid).
  const centroide = await geocodeComunaCentroide(comuna)
  if (centroide.ok && centroide.lat !== undefined && centroide.lng !== undefined) {
    return {
      lat: centroide.lat,
      lng: centroide.lng,
      comuna,
      precision: 'centroide_comuna',
      direccionLabel: centroide.displayName ?? comuna,
      fuenteTexto: comuna,
    }
  }

  return null
}
```

```typescript
import { circle as turfCircle } from '@turf/turf'
import { consultarIsocronaOrs, type IsocronaModo } from '@/lib/isocrona-server'
import type { IsocronaResultado } from '@/lib/cabida-comercial'

function round6(n: number): number {
  return Math.round(n * 1e6) / 1e6
}

// Radios de referencia (metros) para el círculo equivalente cuando ORS falla
// — NO calibrados para Chile, son un piso razonable de caminata/auto a la
// velocidad promedio urbana. Documentado en la UI como método degradado, no
// como isócrona real (Pitfall 1, 16-RESEARCH.md).
const VELOCIDAD_M_POR_MIN = { caminando: 80, auto: 500 } as const

/**
 * Cache-through hacia cabida_comercial_cache, clonando el patrón de
 * app/api/zonificacion/lookup/route.ts (round6, upsert incondicional,
 * force salta la lectura pero NUNCA la escritura). NUNCA lanza, NUNCA
 * retorna una geometría sin decidir metodo explícitamente (Pitfall 1).
 */
export async function obtenerIsocrona(
  ubicacion: Pick<UbicacionCabida, 'lat' | 'lng'>,
  opts: { modo: IsocronaModo; minutos: number; force?: boolean },
): Promise<IsocronaResultado> {
  const supabase = createServiceClient()
  const latR = round6(ubicacion.lat)
  const lngR = round6(ubicacion.lng)

  if (!opts.force) {
    const { data: cached } = await supabase
      .from('cabida_comercial_cache')
      .select('*')
      .eq('lat_r', latR)
      .eq('lng_r', lngR)
      .eq('modo', opts.modo)
      .eq('minutos', opts.minutos)
      .eq('isocrona_status', 'encontrado')
      .maybeSingle()

    if (cached) {
      return {
        metodo: cached.isocrona_metodo as IsocronaResultado['metodo'],
        geometria: cached.isocrona_geometria,
        modo: opts.modo,
        minutos: opts.minutos,
        proveedor: cached.isocrona_proveedor as IsocronaResultado['proveedor'],
        consultadoEl: cached.consultado_el,
        cacheHit: true,
      }
    }
  }

  const orsResult = await consultarIsocronaOrs(ubicacion.lat, ubicacion.lng, opts.modo, opts.minutos)
  const nowIso = new Date().toISOString()

  let resultado: IsocronaResultado
  if (orsResult.ok) {
    resultado = {
      metodo: 'red_vial',
      geometria: orsResult.geometria,
      modo: opts.modo,
      minutos: opts.minutos,
      proveedor: 'openrouteservice',
      consultadoEl: nowIso,
      cacheHit: false,
    }
  } else {
    console.warn(`[cabida-comercial] ORS no disponible (${orsResult.error}) — degradando a círculo equivalente`)
    const radioM = VELOCIDAD_M_POR_MIN[opts.modo] * opts.minutos
    const circulo = turfCircle([ubicacion.lng, ubicacion.lat], radioM / 1000, { units: 'kilometers' })
    resultado = {
      metodo: 'circulo_equivalente',
      geometria: circulo.geometry as IsocronaResultado['geometria'],
      modo: opts.modo,
      minutos: opts.minutos,
      proveedor: null,
      consultadoEl: nowIso,
      cacheHit: false,
    }
  }

  // upsert incondicional — mismo criterio que zonificacion_cache: seguro
  // tanto en cache-miss normal como en force=true.
  const { error: upsertErr } = await supabase.from('cabida_comercial_cache').upsert(
    {
      lat_r: latR,
      lng_r: lngR,
      modo: opts.modo,
      minutos: opts.minutos,
      isocrona_status: 'encontrado',
      isocrona_metodo: resultado.metodo,
      isocrona_geometria: resultado.geometria,
      isocrona_proveedor: resultado.proveedor,
      consultado_el: nowIso,
    },
    { onConflict: 'lat_r,lng_r,modo,minutos' },
  )
  if (upsertErr) {
    console.error('[cabida-comercial] No se pudo cachear la isócrona:', upsertErr.message)
    // Igual que zonificacion/lookup: el resultado ya calculado se retorna
    // aunque la escritura de caché haya fallado — nunca bloquear la
    // respuesta al usuario por un efecto secundario de caching.
  }

  return resultado
}
```

```typescript
import type { AnalisisCabidaComercial, FormatoComercial } from '@/lib/cabida-comercial'

/**
 * CABI-01: función pura por FIRMA (lat, lng, formato) → resultado. No recibe
 * ni necesita oportunidadId — quien la llama (la ruta, Plan 16-05) ya resolvió
 * la ubicación vía resolverUbicacionDesdeOportunidad()/resolverUbicacionDesdeDireccion()
 * ANTES de invocar esta función. `formato` se acepta y se devuelve sin
 * ramificar ningún cálculo todavía (Fase 18 lo usará para
 * competencia-por-formato) — fijar la firma ahora evita un refactor caro
 * después (16-RESEARCH.md, "Motor desacoplado").
 *
 * Nombre reservado a propósito para ESTA función pura server-only — el
 * helper client-safe de lib/cabida-comercial.ts (Plan 16-02) se llama
 * consultarCabidaComercial (objeto-param, hace fetch HTTP), NO
 * obtenerAnalisisCabidaComercial, para evitar la ambigüedad de dos
 * funciones con el mismo nombre y firmas incompatibles — mismo criterio que
 * lookupZonificacion() (client) vs persistZonificacionParaProyecto()
 * (server) en lib/zonificacion.ts / lib/zonificacion-server.ts.
 */
export async function obtenerAnalisisCabidaComercial(
  lat: number,
  lng: number,
  formato: FormatoComercial,
  opts?: { modo?: 'caminando' | 'auto'; minutos?: number; force?: boolean },
): Promise<AnalisisCabidaComercial> {
  const modo = opts?.modo ?? 'caminando'
  const minutos = opts?.minutos ?? 15

  const isocrona = await obtenerIsocrona({ lat, lng }, { modo, minutos, force: opts?.force })

  return {
    formato,
    isocrona,
    generadoEl: new Date().toISOString(),
  }
}
```
