```typescript
import { resolverUbicacionDesdeOportunidad, resolverUbicacionDesdeDireccion, obtenerAnalisisCabidaComercial } from '@/lib/cabida-comercial-server'
import type { CabidaComercialAnalisisResponse, FormatoComercial } from '@/lib/cabida-comercial'
import { checkRateLimit } from '@/lib/rate-limit'

export const dynamic = 'force-dynamic'

const FORMATOS_VALIDOS: FormatoComercial[] = ['supermercado', 'minimarket', 'strip_center', 'power_center']

function esFormatoValido(v: string | null): v is FormatoComercial {
  return v !== null && (FORMATOS_VALIDOS as string[]).includes(v)
}

export async function GET(request: Request): Promise<Response> {
  const { searchParams } = new URL(request.url)
  const oportunidadId = searchParams.get('oportunidadId')
  const direccion = searchParams.get('direccion')
  const comuna = searchParams.get('comuna')
  const formatoParam = searchParams.get('formato')
  const force = searchParams.get('force') === 'true'
  const formato: FormatoComercial = esFormatoValido(formatoParam) ? formatoParam : 'supermercado'
  // Sin selector de formato en la UI todavía (COMPE-01 llega en Fase 18) —
  // 'supermercado' es el default documentado hasta entonces. El parámetro
  // se acepta desde ahora para no tener que tocar esta ruta cuando el
  // selector exista (CABI-01).

  if (!oportunidadId && !(direccion && comuna)) {
    return Response.json(
      { ok: false, error: 'Se requiere "oportunidadId" o "direccion"+"comuna"' } satisfies CabidaComercialAnalisisResponse,
      { status: 400 },
    )
  }

  const ip = request.headers.get('x-forwarded-for')?.split(',')[0]?.trim()
    ?? request.headers.get('x-real-ip')
    ?? 'unknown'
  const rateLimit = await checkRateLimit(`cabida-comercial:${ip}`)
  if (rateLimit) return rateLimit

  try {
    const ubicacion = oportunidadId
      ? await resolverUbicacionDesdeOportunidad(oportunidadId)
      : await resolverUbicacionDesdeDireccion(direccion as string, comuna as string)

    if (!ubicacion) {
      return Response.json(
        { ok: false, error: 'No se pudo resolver la ubicación (geocoding y centroide de comuna fallaron)' } satisfies CabidaComercialAnalisisResponse,
        { status: 502 },
      )
    }

    const analisis = await obtenerAnalisisCabidaComercial(ubicacion.lat, ubicacion.lng, formato, { force })

    return Response.json(
      { ok: true, ubicacion, analisis } satisfies CabidaComercialAnalisisResponse,
      { status: 200 },
    )
  } catch (err) {
    console.error('[cabida-comercial] Error inesperado en /analisis:', err instanceof Error ? err.message : err)
    return Response.json(
      { ok: false, error: 'Error interno al calcular cabida comercial' } satisfies CabidaComercialAnalisisResponse,
      { status: 500 },
    )
  }
}
```

```typescript
"use client"

import { useEffect, useState } from "react"
import { AlertTriangle, Loader2, MapPin, RefreshCw } from "lucide-react"
import { toast } from "sonner"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Button } from "@/components/ui/button"
import { obtenerAnalisisCabidaComercial, type CabidaComercialAnalisisResponse } from "@/lib/cabida-comercial"

interface CabidaComercialTabProps {
  oportunidadId: string
}

const PRECISION_LABEL: Record<string, string> = {
  aproximada: "Dirección aproximada (geocodificada desde el aviso)",
  centroide_comuna: "Centroide de comuna — no se pudo geocodificar una dirección más precisa",
}

export function CabidaComercialTab({ oportunidadId }: CabidaComercialTabProps) {
  const [data, setData] = useState<CabidaComercialAnalisisResponse | null>(null)
  const [loading, setLoading] = useState(true)
  const [actualizando, setActualizando] = useState(false)

  async function cargar(force: boolean) {
    const setter = force ? setActualizando : setLoading
    setter(true)
    try {
      const result = await obtenerAnalisisCabidaComercial({ oportunidadId, force })
      setData(result)
      if (force) {
        if (result.ok) toast.success("Cabida comercial actualizada")
        else toast.error(result.error ?? "No se pudo actualizar")
      }
    } catch {
      if (force) toast.error("No se pudo actualizar")
    } finally {
      setter(false)
    }
  }

  useEffect(() => {
    void cargar(false)
    // Fetch on-demand al montar (mismo criterio que ZonificacionCard) — sin
    // dependencias que disparen refetch en background; el único re-fetch
    // manual es handleActualizar vía el botón (UBIC-05).
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [oportunidadId])

  if (loading) {
    return <p className="text-xs text-muted-foreground">Calculando ubicación y área de influencia…</p>
  }

  if (!data?.ok || !data.ubicacion || !data.analisis) {
    return (
      <div className="rounded-lg border border-red-200 bg-red-50 p-3 text-xs text-red-800">
        <p className="flex items-center gap-1.5 font-medium">
          <AlertTriangle className="size-3.5" /> No se pudo calcular la cabida comercial
        </p>
        <p className="mt-1">{data?.error ?? "Error desconocido"}</p>
      </div>
    )
  }

  const { ubicacion, analisis } = data
  const esDegradado = analisis.isocrona.metodo === "circulo_equivalente"

  return (
    <div className="space-y-4">
      <Card className="rounded-[3px] border-line-strong shadow-none">
        <CardHeader className="flex-row items-center justify-between pb-3">
          <CardTitle className="font-technical text-sm">Ubicación y área de influencia</CardTitle>
          <Button
            variant="ghost"
            size="sm"
            className="h-7 gap-1.5 text-xs"
            onClick={() => void cargar(true)}
            disabled={actualizando}
          >
            {actualizando ? <Loader2 className="size-3.5 animate-spin" /> : <RefreshCw className="size-3.5" />}
            Actualizar
          </Button>
        </CardHeader>
        <CardContent className="space-y-3 text-sm">
          <div>
            <p className="flex items-center gap-1.5 text-[11px] uppercase tracking-wide text-muted-foreground">
              <MapPin className="size-3" /> Ubicación resuelta
            </p>
            <p className="num text-sm text-primary">
              {ubicacion.lat.toFixed(6)}, {ubicacion.lng.toFixed(6)}
            </p>
            <p className="mt-1 text-[11px] text-muted-foreground">{ubicacion.direccionLabel}</p>
            <p className="mt-1 rounded-md border border-line-fine bg-muted/40 px-2 py-1 text-[11px] text-muted-foreground">
              Precisión: {PRECISION_LABEL[ubicacion.precision] ?? ubicacion.precision}
            </p>
          </div>

          <div>
            <p className="text-[11px] uppercase tracking-wide text-muted-foreground">
              Área de influencia ({analisis.isocrona.modo === "caminando" ? "caminata" : "auto"}, {analisis.isocrona.minutos} min)
            </p>
            <p className="mt-1 text-xs text-primary">
              Método: {esDegradado ? "Círculo equivalente" : "Red vial (isócrona real)"}
            </p>
            {esDegradado && (
              <div className="mt-2 rounded-lg border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800">
                <p className="flex items-center gap-1.5 font-medium">
                  <AlertTriangle className="size-3.5" /> Radio aproximado — no se pudo calcular la ruta real
                </p>
                <p className="mt-1">
                  El servicio de ruteo no respondió. Este es un círculo equivalente, no una isócrona real —
                  puede sobre/sub-estimar el área alcanzable según la red vial del sector.
                </p>
              </div>
            )}
            {!esDegradado && (
              <p className="mt-1 text-[11px] text-muted-foreground">
                Calculado vía openrouteservice — polígono de red vial real, no un radio circular.
              </p>
            )}
          </div>
        </CardContent>
      </Card>
    </div>
  )
}
```
