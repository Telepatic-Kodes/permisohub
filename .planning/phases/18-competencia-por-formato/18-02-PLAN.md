```typescript
import * as turf from '@turf/turf'
import type { CompetidorDetectado, FormatoComercial } from '@/lib/cabida-comercial'

const TAG_A_FORMATO: Record<string, FormatoComercial> = {
  supermarket: 'supermercado',
  department_store: 'supermercado',
  mall: 'supermercado',
  convenience: 'minimarket',
}

export async function obtenerCompetidoresOverpass(
  lat: number,
  lng: number,
  isocronaGeometria?: GeoJSON.Polygon | GeoJSON.MultiPolygon | null
): Promise<CompetidorDetectado[]> {
  await throttle()
  // ... consultarOverpass(lat, lng), manejo 429/backoff/OverpassUnavailableError igual que Task 1

  const json = await res.json() as { elements?: OverpassElement[] }
  const origen = turf.point([lng, lat]) // turf usa [lng, lat] — Overpass usa lat,lng. Conversión en ESTE único punto, con comentario inline (mismo pitfall de axis-order ya documentado en 16-RESEARCH.md).

  const competidores: CompetidorDetectado[] = []
  for (const el of json.elements ?? []) {
    const shop = el.tags?.shop
    const formato = shop ? TAG_A_FORMATO[shop] : undefined
    if (!formato) continue

    const elLat = el.lat ?? el.center?.lat
    const elLng = el.lon ?? el.center?.lon
    if (elLat == null || elLng == null) continue

    const punto = turf.point([elLng, elLat])
    if (isocronaGeometria && !turf.booleanPointInPolygon(punto, isocronaGeometria)) continue // fuera del área de influencia real — descartado

    competidores.push({
      nombre: el.tags?.name ?? el.tags?.brand ?? `(sin nombre — ${shop})`,
      formato,
      fuente: 'osm',
      lat: elLat,
      lng: elLng,
      distanciaM: Math.round(turf.distance(origen, punto, { units: 'meters' })),
      confianza: 'media', // corroboración con SII (Plan 18-06) puede subirla a 'alta'; nunca se asume 'alta' solo por venir de OSM
      direccionLabel: undefined,
    })
  }
  return competidores
}
```

Nota: si `isocronaGeometria` es `undefined`/`null` (caller no la pasó — ej. testing standalone antes de que Phase 16 exista), el filtro de polígono se salta y el radio de Overpass (`RADIO_COMPETENCIA_M`) es la única acotación — comportamiento esperado y documentado, no un bug.
