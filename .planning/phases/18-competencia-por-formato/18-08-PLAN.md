```typescript
const FUENTE_LABEL: Record<string, string> = {
  osm: "OSM",
  seed_list: "Lista curada",
  sii_geocodificado: "Nómina SII",
}

function formatearDistancia(m: number): string {
  return m >= 1000 ? `${(m / 1000).toFixed(1)} km` : `${Math.round(m)} m`
}

// ...dentro del JSX, después de la Card de ubicación/isócrona:
<Card className="rounded-[3px] border-line-strong shadow-none">

        <div className="rounded-lg border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800">
          <p className="font-medium">
            {analisis.competencia.competidores.length} competidor
            {analisis.competencia.competidores.length === 1 ? "" : "es"} detectado
            {analisis.competencia.competidores.length === 1 ? "" : "s"} — confianza {analisis.competencia.confianzaGlobal}
          </p>
          <p className="mt-1">{analisis.competencia.disclosure}</p>
        </div>
        {analisis.competencia.competidores.length === 0 ? (
          <p className="text-xs text-muted-foreground">
            Ningún competidor detectado con las fuentes disponibles — ver aviso de cobertura arriba.
          </p>
        ) : (
          <ul className="space-y-1.5">
            {analisis.competencia.competidores.map((c, i) => (
              <li key={i} className="flex items-center justify-between rounded-md border border-line-fine px-2 py-1.5 text-xs">
                <span className="text-primary">{c.nombre}</span>
                <span className="flex items-center gap-2 text-muted-foreground">
                  <span className="rounded bg-muted px-1.5 py-0.5">{FUENTE_LABEL[c.fuente] ?? c.fuente}</span>
                  <span className="num">{formatearDistancia(c.distanciaM)}</span>
                </span>
              </li>
            ))}
          </ul>
        )}
      </>
    )}
  </CardContent>
</Card>
```
