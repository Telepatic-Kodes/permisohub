**2. Sección de metodología (VERE-03) — nuevo componente `cabida-metodologia.tsx`, clonando el PATRÓN visual (no el código, dominio distinto) de `metodologia-informe.tsx`:**

```typescript
// components/mercado-inmobiliario/oportunidad-detalle/cabida-metodologia.tsx
import type { AnalisisCabidaComercial } from "@/lib/cabida-comercial"
import { formatFechaCorta } from "@/lib/formato-fecha"

interface CabidaMetodologiaProps {
  analisis: AnalisisCabidaComercial
}

export function CabidaMetodologia({ analisis }: CabidaMetodologiaProps) {
  const noVerificado: string[] = []
  if (analisis.competencia) noVerificado.push(analisis.competencia.disclosure)
  if (analisis.demografia?.poblacion && !analisis.demografia.poblacion.ok) {
    noVerificado.push("La intersección geoespacial con manzanas censales no fue exitosa para este punto.")
  }
  if (analisis.demografia?.consumo.categoriasPendientes.length) {
    noVerificado.push(`Categorías EPF sin dato: ${analisis.demografia.consumo.categoriasPendientes.join(", ")}.`)
  }
  if (analisis.isocrona.metodo === "circulo_equivalente") {
    noVerificado.push("El área de influencia es un círculo equivalente, no una isócrona de red vial real.")
  }

  return (
    <section className="mt-6 border-t border-line-fine pt-4 text-xs">
      <h3 className="text-[11px] font-semibold uppercase tracking-widest text-muted-foreground">Metodología y fuentes</h3>
      <ul className="mt-2 space-y-1.5 text-muted-foreground">
        {analisis.demografia && (
          <li>
            Población: Censo {analisis.demografia.poblacion.censoAno}, consultado el{" "}
            {formatFechaCorta(analisis.demografia.poblacion.consultadoEl)}.
          </li>
        )}
        {analisis.competencia && (
          <li>Competencia: datos consultados el {formatFechaCorta(analisis.competencia.consultadoEl)}.</li>
        )}
        <li>
          Área de influencia: {analisis.isocrona.modo === "caminando" ? "caminata" : "auto"}, {analisis.isocrona.minutos} min,
          método {analisis.isocrona.metodo === "red_vial" ? "red vial real" : "círculo equivalente"}.
        </li>
      </ul>
      {noVerificado.length > 0 && (
        <div className="mt-3 rounded-lg border border-amber-200 bg-amber-50 p-3 text-amber-800">
          <p className="font-medium">Qué no se pudo verificar</p>
          <ul className="mt-1 space-y-1">
            {noVerificado.map((texto, i) => (
              <li key={i}>{texto}</li>
            ))}
          </ul>
        </div>
      )}
    </section>
  )
}
```
