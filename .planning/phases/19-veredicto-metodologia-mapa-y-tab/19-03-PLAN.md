CREATE OR REPLACE FUNCTION public.calcular_percentiles_gap_score_cabida(p_formato text)
RETURNS TABLE (muestra_n bigint, p33 double precision, p66 double precision)
LANGUAGE sql STABLE
AS $$
  SELECT
    count(*),
    percentile_cont(0.33) WITHIN GROUP (ORDER BY veredicto_gap_score),
    percentile_cont(0.66) WITHIN GROUP (ORDER BY veredicto_gap_score)
  FROM cabida_comercial_cache
  WHERE formato = p_formato AND veredicto_gap_score IS NOT NULL;
$$;

export async function obtenerPercentilesGapScore(formato: FormatoComercial): Promise<PercentilesGapScore | null> {
  const supabase = createServiceClient()
  const { data, error } = await supabase.rpc('calcular_percentiles_gap_score_cabida', { p_formato: formato })
  if (error) throw new Error(`calcular_percentiles_gap_score_cabida falló para formato ${formato}: ${error.message}`)

  return { p33: raw.p33, p66: raw.p66, muestraN: Number(raw.muestra_n) }
}
