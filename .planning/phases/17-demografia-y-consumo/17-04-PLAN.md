<Card className="rounded-[3px] border-line-strong shadow-none">
  <CardHeader className="pb-3">
    <CardTitle className="font-technical text-sm">Demografía y consumo</CardTitle>
  </CardHeader>
  <CardContent className="space-y-4 text-sm">
    {!analisis.demografia ? (
      <p className="text-xs text-muted-foreground">Análisis de demografía no disponible.</p>

              Censo 2017 — INE
            </span>
          </div>

          </p>
        </div>

          <div className="flex items-center justify-between">
            <span className="font-medium">Capacidad de gasto estimada</span>
            <span className="rounded bg-muted px-1.5 py-0.5 text-xs text-muted-foreground">
              EPF {analisis.demografia.consumo.epfAno} · CASEN {analisis.demografia.consumo.casenAno}
            </span>
          </div>
          <div className="rounded-lg border border-amber-200 bg-amber-50 p-3 text-xs text-amber-800">
            {analisis.demografia.consumo.disclosure}
          </div>

            {analisis.demografia.consumo.tasaPobrezaComunal !== null
              ? `Tasa de pobreza comunal: ${analisis.demografia.consumo.tasaPobrezaComunal}% (CASEN ${analisis.demografia.consumo.casenAno})`
              : "Sin dato de pobreza comunal transcrito para esta comuna."}
          </p>
          <ul className="space-y-1">
            {analisis.demografia.consumo.categorias.map((c) => (
              <li key={c.nombre} className="flex items-center justify-between text-xs">
                <span>{c.nombre}</span>
                <span className={c.participacionPct === null ? "text-muted-foreground" : "num"}>
                  {c.participacionPct === null ? "pendiente de transcripción" : `${(c.participacionPct * 100).toFixed(1)}%`}
                </span>
              </li>
            ))}
          </ul>
        </div>
      </>
    )}
  </CardContent>
</Card>
